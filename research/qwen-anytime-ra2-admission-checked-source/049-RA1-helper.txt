def qualification_ids(rows):
    need(type(rows) is list and len(rows)<=30000,'typed_bounded_qualification_known_required')
    keys=[]
    for row in rows:
        need(type(row) is dict and type(row.get('pid')) is int and row['pid']>0
             and type(row.get('create_time')) in (int,float) and math.isfinite(row['create_time'])
             and row['create_time']>0,'finite_exact_qualification_identity_required')
        keys.append((row['pid'],row['create_time']))
    need(len(keys)==len(set(keys)),'unique_qualification_known_required')
    return set(keys)

         and failed.get('actual_exec_session')==49569 and failed.get('actual_exec_exit_code_observed_by_root')==1

         and inherited['common_deadline_monotonic']==inherited['start_monotonic']+6000,

             ==qualification_ids(prior_post['known_identities']) and len(oldknown)==8016
         and prior_post.get('negative_closure_sha256')==expected['negative']['sha256']

    context=bound(root['repair_context']);audit_ref={k:root['independent_audit'][k] for k in ('path','sha256')};audit=bound(audit_ref)

         and context.get('original_failure_preserved') is True and context.get('original_RQ6_procedure_requalified') is False
         and all(context.get(k)==expected[v] for k,v in [('original_manifest','manifest'),('original_campaign','campaign'),
